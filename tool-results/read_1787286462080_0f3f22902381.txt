    94→function getMagColor(mag: number): string {
    95→  if (mag >= 7) return '#dc2626'
    96→  if (mag >= 6) return '#ea580c'
    97→  if (mag >= 5) return '#ca8a04'
    98→  if (mag >= 4) return '#65a30d'
    99→  return '#0891b2'
   100→}
   101→
   102→function getMagLabel(mag: number): string {
   103→  if (mag >= 8) return 'EKSTREM'
   104→  if (mag >= 7) return 'BERAT'
   105→  if (mag >= 6) return 'KUAT'
   106→  if (mag >= 5) return 'SEDANG'
   107→  if (mag >= 4) return 'RINGAN'
   108→  return 'MINOR'
   109→}
   110→
   111→function timeAgo(isoDate: string): string {
   112→  const diff = Date.now() - new Date(isoDate).getTime()
   113→  const mins = Math.floor(diff / 60000)
   114→  if (mins < 1) return 'Baru saja'
   115→  if (mins < 60) return `${mins} menit lalu`
   116→  const hrs = Math.floor(mins / 60)
   117→  if (hrs < 24) return `${hrs} jam lalu`
   118→  const days = Math.floor(hrs / 24)
   119→  return `${days} hari lalu`
   120→}
